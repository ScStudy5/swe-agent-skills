# Plantilla base .gitattributes (normalización LF y protección de binarios)
# Destino recomendado en tu repo: .gitattributes

# Auto-detectar archivos de texto y normalizar a LF al hacer commit
* text=auto eol=lf

# Forzar LF en scripts y configuraciones sensibles a fin de línea (Linux/CI/Docker)
*.sh text eol=lf
*.mjs text eol=lf
*.cjs text eol=lf
*.js text eol=lf
*.ts text eol=lf
*.tsx text eol=lf
*.py text eol=lf
*.yml text eol=lf
*.yaml text eol=lf
*.json text eol=lf
*.md text eol=lf
Dockerfile text eol=lf
Makefile text eol=lf

# Scripts exclusivos de Windows
*.bat text eol=crlf
*.cmd text eol=crlf

*.jpg binary
